💻 Örnek Kod Üretimi

Komut: "Python'da bir dosya okuma ve yazma fonksiyonu yaz"

def read_file(filename):
    """Dosyayı oku ve içeriğini döndür."""
    try:
        with open(filename, 'r', encoding='utf-8') as f:
            return f.read()
    except FileNotFoundError:
        return "Dosya bulunamadı!"

def write_file(filename, content):
    """Dosyaya içerik yaz."""
    try:
        with open(filename, 'w', encoding='utf-8') as f:
            f.write(content)
        return "Dosya yazıldı!"
    except Exception as e:
        return f"Hata: {e}"

# Kullanım
veri = read_file("ornek.txt")
print(veri)
write_file("yeni.txt", "Merhaba VİZYON AI!")
